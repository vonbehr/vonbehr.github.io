{%- comment -%} Renders an with a 1x/2x srcset from manually prepared images named "name.ext" (1x) and "name@2x.ext" (2x). Usage: {% include responsive-image.html src="/assets/images/photo.jpg" alt="A description" %} Optional parameters: suffix - 2x filename suffix, default "@2x" class - CSS class(es) for the width - width attribute height - height attribute {%- endcomment -%} {%- assign parts = include.src | split: "." -%} {%- assign ext = parts | last -%} {%- assign base_count = parts.size | minus: 1 -%} {%- assign base = parts | slice: 0, base_count | join: "." -%} {%- assign suffix = include.suffix | default: "@2x" -%} {%- assign src1x = base | append: "." | append: ext | relative_url -%} {%- assign src2x = base | append: suffix | append: "." | append: ext | relative_url -%}
